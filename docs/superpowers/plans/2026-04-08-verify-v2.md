### Task 3: Create ProtocolFingerprintService (Phase 1)

**Files:**
- Create: `backend/src/verify/tests/protocol-fingerprint.service.ts`

This service detects whether the relay's backend is real Anthropic API vs AWS Bedrock/Kiro vs Google Vertex, using protocol-level fingerprints from [cc-proxy-detector](https://github.com/zxc123aa/cc-proxy-detector).

5 tests: inference_geo field, ratelimit headers, message ID format, tool_use ID prefix, SSE event chain.

**Note:** For OpenAI-format relays, most protocol fingerprints don't apply (they've already been converted). These tests are most valuable for Anthropic-format relays. For OpenAI-format, tests 1.1-1.3 are skipped, test 1.4 is adapted, and test 1.5 checks OpenAI SSE format consistency.

- [ ] **Step 1: Create protocol-fingerprint.service.ts**

```typescript
import { Injectable, Logger } from '@nestjs/common';
import type {
  VerifySession,
  SseEvent,
  TestResult,
  TestStatus,
  TestEvidence,
  EvidenceItem,
} from '../verify.types.js';

@Injectable()
export class ProtocolFingerprintService {
  private readonly logger = new Logger(ProtocolFingerprintService.name);

  async run(
    session: VerifySession,
    emit: (event: SseEvent) => void,
  ): Promise<TestResult[]> {
    const results: TestResult[] = [];

    // Make initial probe request and capture full response for fingerprint analysis
    const probe = await this.probeRequest(session);

    results.push(this.testInferenceGeo(probe, session, emit));
    results.push(this.testRatelimitHeaders(probe, session, emit));
    results.push(this.testMessageIdFormat(probe, session, emit));
    results.push(await this.testToolUsePrefix(session, emit));
    results.push(await this.testSseEventChain(session, emit));

    return results;
  }

  // ── Probe request: capture headers + body ─────────────

  private async probeRequest(session: VerifySession): Promise<{
    headers: Headers;
    body: any;
    status: number;
  } | null> {
    try {
      const baseUrl = session.baseUrl.replace(/\/$/, '');

      if (session.format === 'anthropic') {
        const res = await fetch(`${baseUrl}/v1/messages`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'x-api-key': session.apiKey,
            'anthropic-version': '2023-06-01',
          },
          body: JSON.stringify({
            model: session.model,
            max_tokens: 32,
            messages: [{ role: 'user', content: 'Say "hello"' }],
          }),
          signal: AbortSignal.timeout(30000),
        });
        const body = await res.json().catch(() => null);
        return { headers: res.headers, body, status: res.status };
      } else {
        const res = await fetch(`${baseUrl}/v1/chat/completions`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            Authorization: `Bearer ${session.apiKey}`,
          },
          body: JSON.stringify({
            model: session.model,
            max_tokens: 32,
            messages: [{ role: 'user', content: 'Say "hello"' }],
          }),
          signal: AbortSignal.timeout(30000),
        });
        const body = await res.json().catch(() => null);
        return { headers: res.headers, body, status: res.status };
      }
    } catch (err: any) {
      this.logger.warn(`Probe request failed: ${err.message}`);
      return null;
    }
  }

  // ── helpers ──────────────────────────────────────────

  private makeResult(
    test: number, name: string, status: TestStatus,
    detail: string, duration: number, evidence?: TestEvidence,
  ): TestResult {
    return { phase: 1, test, name, status, detail, duration, evidence };
  }

  // ── Test 1.1: inference_geo field ─────────────────────

  private testInferenceGeo(
    probe: { headers: Headers; body: any; status: number } | null,
    session: VerifySession,
    emit: (event: SseEvent) => void,
  ): TestResult {
    const name = 'inference_geo 字段';
    emit({ type: 'test-start', phase: 1, test: 1, name });
    const t0 = Date.now();

    if (session.format === 'openai') {
      const evidence: TestEvidence = { expected: 'Anthropic 格式专属字段', actual: '当前为 OpenAI 格式，跳过' };
      const result = this.makeResult(1, name, 'skip', 'OpenAI 格式不适用', Date.now() - t0, evidence);
      emit({ type: 'test-result', ...result });
      return result;
    }

    if (!probe || !probe.body) {
      const result = this.makeResult(1, name, 'warn', '探测请求失败', Date.now() - t0);
      emit({ type: 'test-result', ...result });
      return result;
    }

    const hasGeo = 'inference_geo' in probe.body;
    const geoValue = probe.body.inference_geo;

    let status: TestStatus;
    let detail: string;

    if (hasGeo) {
      status = 'pass';
      detail = `检测到 inference_geo: "${geoValue}"，确认为 Anthropic 原生 API`;
    } else {
      status = 'fail';
      detail = '响应中无 inference_geo 字段，非 Anthropic 原生（可能为 Bedrock/Vertex）';
    }

    const evidence: TestEvidence = {
      expected: '存在 inference_geo 字段（Anthropic API 始终包含）',
      actual: hasGeo ? `inference_geo: "${geoValue}"` : '字段不存在',
      raw: JSON.stringify(
        Object.fromEntries(
          Object.entries(probe.body).filter(([k]) =>
            ['id', 'type', 'model', 'inference_geo', 'service_tier'].includes(k)
          )
        ),
        null, 2,
      ),
    };

    const result = this.makeResult(1, name, status, detail, Date.now() - t0, evidence);
    emit({ type: 'test-result', ...result });
    return result;
  }

  // ── Test 1.2: Ratelimit headers ───────────────────────

  private testRatelimitHeaders(
    probe: { headers: Headers; body: any; status: number } | null,
    session: VerifySession,
    emit: (event: SseEvent) => void,
  ): TestResult {
    const name = 'Ratelimit 头验证';
    emit({ type: 'test-start', phase: 1, test: 2, name });
    const t0 = Date.now();

    if (session.format === 'openai') {
      const evidence: TestEvidence = { expected: 'Anthropic 格式专属头部', actual: '当前为 OpenAI 格式，跳过' };
      const result = this.makeResult(2, name, 'skip', 'OpenAI 格式不适用', Date.now() - t0, evidence);
      emit({ type: 'test-result', ...result });
      return result;
    }

    if (!probe) {
      const result = this.makeResult(2, name, 'warn', '探测请求失败', Date.now() - t0);
      emit({ type: 'test-result', ...result });
      return result;
    }

    // Collect all anthropic-ratelimit headers
    const rlHeaders: Record<string, string> = {};
    probe.headers.forEach((value, key) => {
      if (key.startsWith('anthropic-ratelimit')) {
        rlHeaders[key] = value;
      }
    });

    const hasRlHeaders = Object.keys(rlHeaders).length > 0;
    const hasUnified5h = Object.keys(rlHeaders).some((k) => k.includes('unified-5h'));

    let status: TestStatus;
    let detail: string;

    if (hasRlHeaders && hasUnified5h) {
      status = 'pass';
      detail = '检测到完整 ratelimit 头（含 unified-5h），确认 Anthropic 原生';
    } else if (hasRlHeaders) {
      status = 'pass';
      detail = `检测到 ${Object.keys(rlHeaders).length} 个 ratelimit 头`;
    } else {
      status = 'fail';
      detail = '未检测到 anthropic-ratelimit-* 头，非 Anthropic 原生';
    }

    const evidence: TestEvidence = {
      expected: '存在 anthropic-ratelimit-* 响应头（含递减的 remaining 值）',
      actual: hasRlHeaders
        ? `检测到 ${Object.keys(rlHeaders).length} 个头部`
        : '无 ratelimit 头',
      raw: hasRlHeaders
        ? Object.entries(rlHeaders).map(([k, v]) => `${k}: ${v}`).join('\n')
        : undefined,
    };

    const result = this.makeResult(2, name, status, detail, Date.now() - t0, evidence);
    emit({ type: 'test-result', ...result });
    return result;
  }

  // ── Test 1.3: Message ID format ───────────────────────

  private testMessageIdFormat(
    probe: { headers: Headers; body: any; status: number } | null,
    session: VerifySession,
    emit: (event: SseEvent) => void,
  ): TestResult {
    const name = 'Message ID 格式';
    emit({ type: 'test-start', phase: 1, test: 3, name });
    const t0 = Date.now();

    if (session.format === 'openai') {
      const evidence: TestEvidence = { expected: 'Anthropic 格式专属字段', actual: '当前为 OpenAI 格式，跳过' };
      const result = this.makeResult(3, name, 'skip', 'OpenAI 格式不适用', Date.now() - t0, evidence);
      emit({ type: 'test-result', ...result });
      return result;
    }

    if (!probe || !probe.body) {
      const result = this.makeResult(3, name, 'warn', '探测请求失败', Date.now() - t0);
      emit({ type: 'test-result', ...result });
      return result;
    }

    const id: string = probe.body.id ?? '';

    let status: TestStatus;
    let detail: string;
    let source = 'unknown';

    if (/^msg_[A-Za-z0-9]+$/.test(id) && !id.includes('-')) {
      // Anthropic native: msg_<base62>, no hyphens
      status = 'pass';
      detail = `ID 格式 msg_<base62>（无连字符），Anthropic 原生`;
      source = 'Anthropic';
    } else if (id.startsWith('msg_') && id.includes('-')) {
      // Bedrock: msg_<UUID> with hyphens
      status = 'fail';
      detail = `ID 含连字符（UUID 格式），疑似 AWS Bedrock`;
      source = 'Bedrock';
    } else if (id.startsWith('req_vrtx_')) {
      status = 'fail';
      detail = `ID 前缀 req_vrtx_，确认为 Google Vertex`;
      source = 'Vertex';
    } else if (id.startsWith('msg_msg_')) {
      status = 'fail';
      detail = `ID 双前缀 msg_msg_，中转站伪装穿帮`;
      source = '伪装失败';
    } else {
      status = 'warn';
      detail = `ID 格式未识别: ${id.slice(0, 30)}`;
    }

    const evidence: TestEvidence = {
      expected: 'msg_<base62>（无连字符，Anthropic 原生格式）',
      actual: `"${id.slice(0, 50)}"${source !== 'unknown' ? ` → ${source}` : ''}`,
      raw: `id: "${id}"`,
    };

    const result = this.makeResult(3, name, status, detail, Date.now() - t0, evidence);
    emit({ type: 'test-result', ...result });
    return result;
  }

  // ── Test 1.4: Tool Use ID prefix ──────────────────────

      if (session.format === 'anthropic') {

            'Content-Type': 'application/json',

              name: 'get_weather',
              description: 'Get current weather for a location',
              input_schema: { type: 'object', properties: { location: { type: 'string' } }, required: ['location'] },

            tool_choice: { type: 'any' },

        const toolBlock = (json.content ?? []).find((b: any) => b.type === 'tool_use');

        if (!toolBlock) {
          const evidence: TestEvidence = { expected: '返回 tool_use 内容块', actual: '未触发 tool_use' };

        } else {
          status = 'warn';
          detail = `ID 格式未识别: ${toolId.slice(0, 30)}`;

        const toolCall = json.choices?.[0]?.message?.tool_calls?.[0];

          const result = this.makeResult(4, name, 'warn', '未触发 tool call，无法检测 ID', Date.now() - t0, evidence);
          emit({ type: 'test-result', ...result });
          return result;
        }

        const callId: string = toolCall.id ?? '';
        const evidence: TestEvidence = {
          expected: 'OpenAI 格式 tool call ID',
          actual: `"${callId.slice(0, 50)}"`,
          raw: `tool_calls[0].id: "${callId}"`,
        };

        const result = this.makeResult(4, name, 'pass', `Tool call ID: ${callId.slice(0, 30)}`, Date.now() - t0, evidence);
        emit({ type: 'test-result', ...result });
        return result;
      }
    } catch (err: any) {
      const result = this.makeResult(4, name, 'warn', `API 调用失败: ${err.message ?? err}`, Date.now() - t0);
      emit({ type: 'test-result', ...result });
      return result;
    }
  }

  // ── Test 1.5: SSE Event Chain ─────────────────────────

  private async testSseEventChain(
    session: VerifySession,
    emit: (event: SseEvent) => void,
  ): Promise<TestResult> {
    const name = 'SSE 事件链';
    emit({ type: 'test-start', phase: 1, test: 5, name });
    const t0 = Date.now();

    try {
      const baseUrl = session.baseUrl.replace(/\/$/, '');
      let res: Response;

      if (session.format === 'anthropic') {
        res = await fetch(`${baseUrl}/v1/messages`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'x-api-key': session.apiKey,
            'anthropic-version': '2023-06-01',
          },
          body: JSON.stringify({
            model: session.model,
            max_tokens: 64,
            stream: true,
            messages: [{ role: 'user', content: 'Hi' }],
          }),
          signal: AbortSignal.timeout(30000),
        });
      } else {
        res = await fetch(`${baseUrl}/v1/chat/completions`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            Authorization: `Bearer ${session.apiKey}`,
          },
          body: JSON.stringify({
            model: session.model,
            max_tokens: 64,
            stream: true,
            messages: [{ role: 'user', content: 'Hi' }],
          }),
          signal: AbortSignal.timeout(30000),
        });
      }

      if (!res.ok || !res.body) {
        const result = this.makeResult(5, name, 'warn', `流式请求失败: HTTP ${res.status}`, Date.now() - t0);
        emit({ type: 'test-result', ...result });
        return result;
      }

      // Read SSE events from the stream
      const reader = res.body.getReader();
      const decoder = new TextDecoder();
      const eventTypes: string[] = [];
      let buffer = '';

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        buffer += decoder.decode(value, { stream: true });

        // Parse SSE lines
        const lines = buffer.split('\n');
        buffer = lines.pop() ?? '';

  private async callApi(
    session: VerifySession,
    messages: { role: string; content: string }[],
    options?: { maxTokens?: number; thinking?: boolean },
  ): Promise<ApiResponse> {
    const maxRetries = 2;
    let lastErr: Error | undefined;

    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        const baseUrl = session.baseUrl.replace(/\/$/, '');
        const maxTokens = options?.maxTokens ?? 256;
        const thinking = options?.thinking ?? false;

        if (session.format === 'anthropic') {
          return await this.callAnthropic(baseUrl, session.apiKey, session.model, messages, maxTokens, thinking);
        }
        return await this.callOpenAI(baseUrl, session.apiKey, session.model, messages, maxTokens);
      } catch (err: any) {
        lastErr = err;
        const is5xx = /\b5\d{2}\b/.test(err.message || '');
        if (is5xx && attempt < maxRetries) {
          this.logger.warn(`API call failed (attempt ${attempt + 1}), retrying in 2s: ${err.message}`);
          await new Promise((r) => setTimeout(r, 2000));
          continue;
        }
        throw err;
      }
    }
    throw lastErr;
  }

  private async callAnthropic(
    baseUrl: string, apiKey: string, model: string,
    messages: { role: string; content: string }[],
    maxTokens: number, thinking: boolean,
  ): Promise<ApiResponse> {
    const body: any = {
      model,
      max_tokens: thinking ? 4096 : maxTokens,
      messages,
    };
    if (thinking) {
      body.thinking = { type: 'enabled', budget_tokens: 1024 };
    }

    const res = await fetch(`${baseUrl}/v1/messages`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': apiKey,
        'anthropic-version': '2023-06-01',
      },
      body: JSON.stringify(body),
      signal: AbortSignal.timeout(30000),
    });

    if (!res.ok) {
      const text = await res.text().catch(() => '');
      throw new Error(`Anthropic API ${res.status}: ${text.slice(0, 200)}`);
    }

    const json = (await res.json()) as any;
    const contentBlocks: any[] = json.content ?? [];

    const textContent = contentBlocks
      .filter((b: any) => b.type === 'text')
      .map((b: any) => b.text)
      .join('');

    const thinkingBlock = contentBlocks.find((b: any) => b.type === 'thinking');
    const thinkingSignature = thinkingBlock?.thinking ?? undefined;

    return { content: textContent, thinkingSignature };
  }

    baseUrl: string, apiKey: string, model: string,
    messages: { role: string; content: string }[],
    maxTokens: number,
  ): Promise<ApiResponse> {

      method: 'POST',
      headers: {

        Authorization: `Bearer ${apiKey}`,

      body: JSON.stringify({ model, messages, max_tokens: maxTokens }),

    });

      const text = await res.text().catch(() => '');
      throw new Error(`OpenAI API ${res.status}: ${text.slice(0, 200)}`);
    }

    const json = (await res.json()) as any;
    return { content: json.choices?.[0]?.message?.content ?? '' };
  }

const SESSION_TTL = 600;

@Injectable()
export class VerifyService {
  private readonly logger = new Logger(VerifyService.name);
  private sessions = new Map<string, Subject<SseEvent>>();

  constructor(
    @InjectRedis() private readonly redis: Redis,
    @InjectRepository(VerifyResult)

  async createSession(dto: StartVerifyDto): Promise<VerifySession> {
    const id = uuidv7();
    const format = await this.detectFormat(dto.baseUrl, dto.apiKey, dto.model);
    const session: VerifySession = {
      id,
      baseUrl: dto.baseUrl,

      mode: dto.mode,
      format,
      createdAt: Date.now(),
    };

    await this.redis.set(
      `verify:${id}`,
      JSON.stringify(session),
      'EX',
      SESSION_TTL,
    );

    this.sessions.set(id, new Subject<SseEvent>());

    return session;
  }

  getEventStream(sessionId: string): Subject<SseEvent> | undefined {
    return this.sessions.get(sessionId);
  }

  async getSession(sessionId: string): Promise<VerifySession | null> {
    const data = await this.redis.get(`verify:${sessionId}`);
    if (!data) return null;
    return JSON.parse(data) as VerifySession;
  }

  async deleteSession(sessionId: string): Promise<void> {
    await this.redis.del(`verify:${sessionId}`);
    const subject = this.sessions.get(sessionId);
    if (subject) {
      subject.complete();
      this.sessions.delete(sessionId);
    }
  }

  emitEvent(sessionId: string, event: SseEvent): void {
    const subject = this.sessions.get(sessionId);
    if (subject) {
      subject.next(event);
    }
  }

    const entity = this.verifyResultRepo.create(data);

    return this.verifyResultRepo.find({

  }

  async getLeaderboardStats(): Promise<{

    baseUrl: string,
    apiKey: string,
    model: string,

    const base = baseUrl.replace(/\/$/, '');

    try {

        method: 'POST',

          'Content-Type': 'application/json',
          'x-api-key': apiKey,
          'anthropic-version': '2023-06-01',

          max_tokens: 16,
          messages: [{ role: 'user', content: 'hi' }],
        }),
        signal: AbortSignal.timeout(15000),
      });
      if (res.ok) {
        this.logger.log(`Format detected: anthropic (${base})`);
        return 'anthropic';
      }
      const text = await res.text().catch(() => '');
      if (text.includes('"type":"error"') && text.includes('"message"')) {
        this.logger.log(`Format detected: anthropic (error response matches Anthropic schema)`);
        return 'anthropic';
      }
    } catch {
      // timeout or network error, try OpenAI
    }

    try {
      const res = await fetch(`${base}/v1/chat/completions`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${apiKey}`,
        },

          model,
          max_tokens: 16,
          messages: [{ role: 'user', content: 'hi' }],
        }),
        signal: AbortSignal.timeout(15000),
      });
      if (res.ok || res.status < 500) {
        this.logger.log(`Format detected: openai (${base})`);
        return 'openai';
      }
    } catch {
      // fallback
    }

    this.logger.warn(`Format detection failed for ${base}, defaulting to openai`);
    return 'openai';

    if (!session) {
      this.logger.warn(`Session ${sessionId} not found, aborting tests`);
      return;
    }

    const emit = (event: SseEvent) => this.emitEvent(sessionId, event);

    // Phase 1: Protocol Fingerprint, Phase 2: Relay Manipulation, Phase 3: Extraction, Phase 4: Model Fingerprint
    const phases = [
      { phase: 1, service: this.protocolFingerprint },
      { phase: 2, service: this.relayManipulation },
      { phase: 3, service: this.extraction },
      { phase: 4, service: this.modelFingerprint },
    ];

    // Quick mode skips Phase 3 (extraction & jailbreak — 6 API calls)
    const activePhases = session.mode === 'quick'
      ? phases.filter((p) => p.phase !== 3)
      : phases;

    const allResults: TestResult[] = [];
    const totals = { pass: 0, warn: 0, fail: 0, skip: 0 };

    for (const { phase, service } of activePhases) {
      const results = await service.run(session, emit);
      allResults.push(...results);

      const summary: PhaseSummary = { phase, pass: 0, warn: 0, fail: 0, skip: 0 };
      for (const r of results) {
        summary[r.status]++;
        totals[r.status]++;
      }

      emit({ type: 'phase-done', phase, summary });
    }

    // Compute verdict with weighted scoring
    const totalTests = totals.pass + totals.warn + totals.fail + totals.skip;
    const failRate = totalTests > 0 ? totals.fail / totalTests : 0;

    let verdict: Verdict;
    let confidence: number;

    if (failRate > 0.3) {
      verdict = 'fake';
      confidence = Math.round(70 + failRate * 30);
    } else if (failRate > 0 || totals.warn > 2) {
      verdict = 'suspicious';
      confidence = Math.round(50 + failRate * 50);
    } else {
      verdict = 'real';
      confidence = 95;
    }

    emit({
      type: 'complete',
      verdict,
      confidence,
      stats: totals,
      results: allResults,
    });

    await this.deleteSession(sessionId);
  }
}
```

                {isExpanded && (
                  <motion.div
                    initial={{ height: 0, opacity: 0 }}
                    animate={{ height: "auto", opacity: 1 }}
                    exit={{ height: 0, opacity: 0 }}
                    transition={{ duration: 0.25 }}
                    className="overflow-hidden"
                  >
                    <div className="px-3 pb-3 space-y-1">
                      {phaseTests.map((test) => {
                        const sCfg = statusConfig[test.status];
                        const testKey = `${test.phase}-${test.test}`;
                        const isTestExpanded = expandedTest === testKey;
                        const hasEvidence = !!test.evidence;

                        return (
                          <div key={testKey}>
                            <div
                              className={`flex items-center justify-between px-3 py-2 rounded-lg ${sCfg.bg} ${
                                hasEvidence ? "cursor-pointer hover:opacity-80" : ""
                              }`}
                              onClick={() => hasEvidence && toggleTest(testKey)}
                            >
                              <div className="flex items-center gap-2.5 min-w-0">
                                <span className={`text-[13px] font-mono w-4 text-center flex-shrink-0 ${sCfg.icon}`}>
                                  {sCfg.symbol}
                                </span>
                                <span className="text-[13px] text-text truncate">{test.name}</span>
                              </div>
                              <div className="flex items-center gap-3 flex-shrink-0 ml-3">
                                {test.detail && (
                                  <span className="text-[11px] text-text-muted max-w-[180px] truncate">
                                    {test.detail}
                                  </span>
                                )}
                                {test.duration != null && (
                                  <span className="text-[11px] font-mono text-text-faint">
                                    {test.duration}ms
                                  </span>
                                )}
                                {hasEvidence && (
                                  <span className={`text-[10px] text-text-faint transition-transform ${isTestExpanded ? "rotate-180" : ""}`}>
                                    {"\u25bc"}
                                  </span>
                                )}
                              </div>
                            </div>

                            <AnimatePresence>
                              {isTestExpanded && test.evidence && (
                                <EvidenceCard evidence={test.evidence} status={test.status} />
                              )}
                            </AnimatePresence>
                          </div>
                        );
                      })}
                    </div>
                  </motion.div>
                )}
              </AnimatePresence>
            </div>
          );
        })}
      </div>

      {/* Action buttons */}
      <div className="flex gap-3 pt-1">
        <button
          onClick={onReset}
          className="flex-1 py-2.5 bg-dark text-bg text-[13px] rounded-lg hover:opacity-90 transition-opacity cursor-pointer"
        >
          {t.verify.report.rerun}
        </button>
        <button className="flex-1 py-2.5 border border-brand/40 text-text-secondary text-[13px] rounded-lg hover:border-brand transition-colors cursor-pointer">
          {t.verify.report.submitToBoard}
        </button>
      </div>
    </div>
  );
}
